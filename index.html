} if(male){ put(root,new THREE.BoxGeometry(.9,.3,.4),shirt,0,1.63,0); } // Hair cap put(root,new THREE.SphereGeometry(.335,16,12),hair,0,2.22,-.055,1.04,.67,1.02); if(female){ // Long hair down the back and both sides put(root,new THREE.BoxGeometry(.58,.9,.16),hair,0,1.78,-.22); parts[side<0?'legL':'legR']=leg; } return {root,parts,type}; } // Floating labels above the mannequins function makeLabel(words,color){ const c=document.createElement('canvas'); c.width=512;c.height=128; ctx.fillText(words,256,65); const texture=new THREE.CanvasTexture(c); const sprite=new THREE.Sprite(new THREE.SpriteMaterial({map:texture,transparent:true,depthTest:false})); sprite.scale.set(3.3,.82,1); return sprite; } // Build the two selectable mannequins in the world const mannequins=[]; function createMannequin(type,x){ const platform=new THREE.Group(); platform.position.set(x,0,-5); scene.add(platform); scene.add(player.root); player.root.position.set(0,0,5); let playerName='Friend'; let playing=false; let selectedType='neutral'; document.querySelector('#startButton').addEventListener('click',()=>{ playerName=document.querySelector('#name').value.trim().slice(0,16)||'Friend'; const dx=e.clientX-lastX; const dy=e.clientY-lastY; if(Math.abs(dx)+Math.abs(dy)>4)dragMoved=true; yaw+=dx*.006; pitch=THREE.MathUtils.clamp(pitch+dy*.004,-.2,1.05); lastX=e.clientX; lastY=e.clientY; }); renderer.domElement.addEventListener('pointerup',e=>{ if(dragging&&!dragMoved&&playing)chooseMannequin(e); activePointers.delete(e.pointerId); dragging=activePointers.size===1; if(dragging){ lastX=e.clientX; lastY=e.clientY; } }); renderer.domElement.addEventListener('pointercancel',e=>{ activePointers.delete(e.pointerId); dragging=false; }); // PC mouse wheel zoom renderer.domElement.addEventListener('wheel',e=>{ distance=THREE.MathUtils.clamp( distance+e.deltaY*.018,minDistance,maxDistance ); },{passive:true}); // Mobile two-finger pinch zoom let pinchStart=0; renderer.domElement.addEventListener('touchstart',e=>{ if(e.touches.length===2){ pinchStart=Math.hypot( e.touches[0].clientX-e.touches[1].clientX, e.touches[0].clientY-e.touches[1].clientY ); } },{passive:true}); renderer.domElement.addEventListener('touchmove',e=>{ if(e.touches.length===2&&pinchStart){ const current=Math.hypot( e.touches[0].clientX-e.touches[1].clientX, e.touches[0].clientY-e.touches[1].clientY ); distance=THREE.MathUtils.clamp( distance+(pinchStart-current)*.035, minDistance,maxDistance ); pinchStart=current; } },{passive:true}); // Keyboard movement const keys={}; addEventListener('keydown',e=>{ keys[e.key.toLowerCase()]=true; if(e.code==='Space'){ e.preventDefault(); jump(); } }); addEventListener('keyup',e=>keys[e.key.toLowerCase()]=false); // Mobile joystick const stick=document.querySelector('#joystick'); const knob=document.querySelector('#knob'); let joyX=0,joyY=0,stickOn=false; function moveStick(e){ const r=stick.getBoundingClientRect(); const dx=e.clientX-(r.left+r.width/2); const dy=e.clientY-(r.top+r.height/2); const len=Math.hypot(dx,dy); const k=Math.min(1,42/(len||1)); joyX=dx*k/42; joyY=dy*k/42; knob.style.left=(50+joyX*38)+'%'; knob.style.top=(50+joyY*38)+'%'; } stick.addEventListener('pointerdown',e=>{ stickOn=true; stick.setPointerCapture(e.pointerId); moveStick(e); }); stick.addEventListener('pointermove',e=>{ if(stickOn)moveStick(e); }); function resetStick(){ stickOn=false; joyX=0; joyY=0; knob.style.left='50%'; knob.style.top='50%'; } stick.addEventListener('pointerup',resetStick); stick.addEventListener('pointercancel',resetStick); // Jump for keyboard and mobile button let vertical=0; let grounded=true; function jump(){ if(playing&&grounded){ vertical=7.3; grounded=false; } } document.querySelector('#jump').addEventListener('pointerdown',e=>{ e.preventDefault(); } // Coordinated walk cycle: // Both arms move together. // Both legs move together, opposite to the arms. const stride=moving?Math.sin(time*(keys.shift?.014:.010))*.55:0; player.parts.armL.rotation.x=stride; player.parts.armR.rotation.x=stride; player.parts.legL.rotation.x=-stride; player.parts.legR.rotation.x=-stride; // Gentle body movement while walking player.parts.torso.position.y=1.28+(moving?Math.abs(stride)*.035:0); } // Gentle idle movement on mannequin platforms mannequins.forEach(m=>{ m.character.root.position.y=.3+Math.sin(time*.0018)*.035; }); updateCamera(); renderer.render(scene,camera); } updateCamera();
